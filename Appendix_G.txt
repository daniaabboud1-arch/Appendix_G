import numpy as np
import matplotlib.pyplot as plt

def mutual_information(p, t):
    s = 1 - p

    terms = [
        (s**2 + t) * np.log2((s**2 + t) / s**2),
        2 * (p*s - t) * np.log2((p*s - t) / (p*s)),
        (p**2 + t) * np.log2((p**2 + t) / p**2)
    ]

    return sum(terms)


def natural_fidelity(p, t):
    s = 1 - p

    return (
        s * np.sqrt(s**2 + t)
        + 2 * np.sqrt(p*s) * np.sqrt(p*s - t)
        + p * np.sqrt(p**2 + t)
    )


p = 0.1

# Small-t values around the Markov point
t_values = np.linspace(-1e-3, 1e-3, 400)

I_values = []
F_nat_values = []

for t in t_values:

    # Keep only admissible points
    if (
        -min(p**2, (1-p)**2) <= t <= p*(1-p)
    ):
        I = mutual_information(p, t)
        F_nat = natural_fidelity(p, t)

        I_values.append(I)
        F_nat_values.append(F_nat)

I_values = np.array(I_values)
F_nat_values = np.array(F_nat_values)

# Exact natural-recovery infidelity
y_exact = 1 - F_nat_values

# Asymptotic prediction
y_asymptotic = (np.log(2) / 4) * I_values

# Sort according to I for plotting
order = np.argsort(I_values)

I_values = I_values[order]
y_exact = y_exact[order]
y_asymptotic = y_asymptotic[order]

plt.figure(figsize=(7, 5))

plt.plot(
    I_values,
    y_exact,
    label=r'Exact $1-F_{\mathrm{nat}}$'
)

plt.plot(
    I_values,
    y_asymptotic,
    '--',
    label=r'Asymptotic $\frac{\ln 2}{4}I(C:R|B)$'
)

plt.xlabel(r'$I(C:R|B)$ (bits)')
plt.ylabel(r'$1-F_{\mathrm{nat}}$')
plt.title('Near-Markov scaling: natural-recovery infidelity')

plt.legend()
plt.tight_layout()

plt.savefig(
    'Figure_6_1.pdf',
    bbox_inches='tight'
)

plt.show()
